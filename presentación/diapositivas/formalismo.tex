\begin{frame}{Notación de Dirac}

	Utilizaremos el formalismo de Dirac:
	
    \begin{columns}[c]
        \column{0.3\textwidth}
        \[
			\ket{\psi} \in \mathcal{H}
        \]

        \column{0.3\textwidth}
        \[
            \bra{\phi} \in \mathcal{H}^*
        \]

        \column{0.4\textwidth}
        \[
			\bra{\phi}  \psi \rangle = \bra{\phi} \left( \ket{\psi} \right) \in \mathbb{C}
        \]
    \end{columns}

	Operador Hermítico:
	\[
		\hat{A}= \hat{A}^\dagger \quad \text{	, tal que } \braket{\phi}{\hat{A} \psi} = \braket{ \hat{A}\phi}{\psi} 
	\]
	
	Como consecuencia, si tenemos la ecuación de autovalores
	\[
		\hat{A}\ket\phi = a \ket\phi \rightarrow a\in\mathbb{R}
	\]
	

\end{frame}
